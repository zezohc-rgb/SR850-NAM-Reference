cmake_minimum_required(VERSION 3.22)
project(SR850NAMReference VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(IR_FILE "${CMAKE_CURRENT_BINARY_DIR}/generated/sr850_minphase.wav")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/make_ir.py" "${IR_FILE}" RESULT_VARIABLE IR_RESULT)
if(NOT IR_RESULT EQUAL 0)
 message(FATAL_ERROR "IR generation failed")
endif()
set(KZ_IR_FILE "${CMAKE_CURRENT_BINARY_DIR}/generated/kz_as16_pro_provisional.wav")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/make_kz_provisional_ir.py" "${KZ_IR_FILE}" RESULT_VARIABLE KZ_RESULT)
if(NOT KZ_RESULT EQUAL 0)
 message(FATAL_ERROR "KZ FIR generation or validation failed")
endif()
set(ART_FILE "${CMAKE_CURRENT_SOURCE_DIR}/sr850_gui.png")
if(EXISTS "${ART_FILE}")
 juce_add_binary_data(SR850IR SOURCES "${IR_FILE}" "${KZ_IR_FILE}" "${ART_FILE}")
 set(SR850_HAS_ARTWORK 1)
else()
 juce_add_binary_data(SR850IR SOURCES "${IR_FILE}" "${KZ_IR_FILE}")
 set(SR850_HAS_ARTWORK 0)
endif()
juce_add_plugin(SR850NAMReference COMPANY_NAME "SR850 Reference" BUNDLE_ID "audio.sr850.namreference" PLUGIN_MANUFACTURER_CODE SrRf PLUGIN_CODE S8Nr FORMATS VST3 PRODUCT_NAME "SR850 NAM Reference" COPY_PLUGIN_AFTER_BUILD OFF)
target_sources(SR850NAMReference PRIVATE Source/PluginProcessor.cpp)
target_link_libraries(SR850NAMReference PRIVATE juce::juce_audio_utils juce::juce_dsp SR850IR)
target_compile_definitions(SR850NAMReference PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)

target_compile_definitions(SR850NAMReference PRIVATE SR850_HAS_ARTWORK=${SR850_HAS_ARTWORK})
